  <script>
    // Scroll animations (Intersection Observer)
    const observer = new IntersectionObserver((entries) => {
      entries.forEach(e => {
        if (e.isIntersecting) {
          e.target.classList.add('visible');
          observer.unobserve(e.target);
        }
      });
    }, { threshold: 0.1 });
    document.querySelectorAll('.fade-in').forEach(el => observer.observe(el));

    // Stagger delays
    document.querySelectorAll('.row .fade-in').forEach((el, i) => {
      el.style.transitionDelay = `${(i % 4) * 0.1}s`;
    });

    // Toast notification
    function showToast(msg) {
      const t = document.getElementById('toast');
      t.textContent = msg;
      t.style.display = 'block';
      setTimeout(() => { t.style.display = 'none'; }, 3500);
    }

    // Contact form
    document.getElementById('contactForm').addEventListener('submit', function(e) {
      e.preventDefault();
      const name = document.getElementById('contact-name').value.trim();
      const email = document.getElementById('contact-email').value.trim();
      const obj = document.getElementById('contact-objective').value;
      const privacy = document.getElementById('privacy-check').checked;
      if (!name || !email || !obj || !privacy) {
        showToast('Por favor completa los campos obligatorios ⚠️');
        return;
      }
      showToast(`¡Gracias ${name}! Te contactamos pronto 🎯`);
      this.reset();
    });

    // Download form
    document.getElementById('downloadForm').addEventListener('submit', function(e) {
      e.preventDefault();
      const name = document.getElementById('download-name').value.trim();
      const email = document.getElementById('download-email').value.trim();
      if (!name || !email) {
        showToast('Completa tu nombre y email ⚠️');
        return;
      }
      showToast(`¡Listo ${name}! Revisa tu email 📩`);
      this.reset();
    });

    // Active nav on scroll
    const sections = document.querySelectorAll('section[id], header[id]');
    const navLinks = document.querySelectorAll('.nav-link[href^="#"]');
    window.addEventListener('scroll', () => {
      let current = '';
      sections.forEach(s => {
        if (window.scrollY >= s.offsetTop - 100) current = s.id;
      });
      navLinks.forEach(a => {
        a.classList.toggle('active', a.getAttribute('href') === '#' + current);
      });
    });
  </script>
